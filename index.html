let selectedColor = '#ff7675'; let notes = JSON.parse(localStorage.getItem('myGroupNotes')) || []; // Inizializza i post-it al caricamento della pagina document.addEventListener("DOMContentLoaded", () => { renderNotes(); }); // Gestione della scelta del colore document.getElementById('colorPicker').addEventListener('click', (e) => { if (e.target.classList.contains('color-option')) { document.querySelectorAll('.color-option').forEach(opt => opt.classList.remove('selected')); e.target.classList.add('selected'); selectedColor = e.target.dataset.color; } }); // Cambia il tema della pagina function changeTheme(themeName) { const body = document.getElementById('mainBody'); body.className = themeName; } // Aggiunge una nuova nota function addNote() { const text = document.getElementById('noteText').value.trim(); if (!text) return alert("Please write something first!"); const newNote = { id: Date.now(), text: text, color: selectedColor, date: new Date().toLocaleDateString('en-GB'), rotation: (Math.random() * 6 - 3).toFixed(1) + 'deg' }; notes.push(newNote); saveAndRender(); document.getElementById('noteText').value = ''; } // Elimina una nota tramite ID function deleteNote(id) { notes = notes.filter(note => note.id !== id); saveAndRender(); } // Salva nel localStorage e aggiorna la vista function saveAndRender() { localStorage.setItem('myGroupNotes', JSON.stringify(notes)); renderNotes(); } // Mostra le note a schermo function renderNotes() { const grid = document.getElementById('notesGrid'); grid.innerHTML = ''; notes.forEach(note => { const noteEl = document.createElement('div'); noteEl.className = 'note'; noteEl.style.backgroundColor = note.color; noteEl.style.setProperty('--rotation', note.rotation); // Corretto l'uso dei backtick per l'HTML multilinea noteEl.innerHTML = `
${note.text}
${note.date}
`; grid.appendChild(noteEl); }); }
